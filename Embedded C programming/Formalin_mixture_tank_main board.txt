void loop() {
  tempRead();
  phfunction();
  distancemesur();
  show_mydata();
  
// ----- Drain Valve Close -----
  if (TankLiquidLevel < 3) {// no liq
    Serial.println("Tank Liquid < 2");

    mixerMotor = false;
    digitalWrite(DRAIN_VALVE, HIGH);
    drainValve = false;
    show_mydata();
	  delay(1000);
    }

   if (TankLiquidLevel <= 6 && !drainValve) {  //0---2
    digitalWrite(MIXER_MOTOR1, HIGH);
    mixerMotor = false;
    digitalWrite(WATER_VALVE, LOW);
    waterValve = true;
    show_mydata();
	  delay(100);
    }
    if (TankLiquidLevel > 18 ) {   //19<
    digitalWrite(WATER_VALVE, HIGH);
    waterValve = false;
    //needdrain=1;
    show_mydata();
    delay(100);
    }
   Serial.print("NBF sensor: ");
   Serial.println(analogRead(NBF_Liquid_Level));
  if (analogRead(NBF_Liquid_Level)<600){   
    Serial.println("NBF sensor ditected");
    digitalWrite(WATER_VALVE, HIGH);
    digitalWrite(NBT_MOTOR, LOW);    
    NBFMotor = true;
    show_mydata();
    for(byte tt=0;tt<60;tt++){
    delay(1000);
    }
    digitalWrite(NBT_MOTOR, HIGH);
    NBFMotor = false;
    show_mydata();
    delay(1000);
  }
  else{
    digitalWrite(NBT_MOTOR, HIGH);
    NBFMotor = false;
    delay(100);
  }
      

  Serial.print("Over flow sensor: ");
   Serial.println(analogRead(Water_Liquid_Level));
   if (analogRead(Water_Liquid_Level)>600){ //this for magnet sensor; if used optical used (wll<800);   
    Serial.println("Over flow sensor ditected");
	  digitalWrite(WATER_VALVE, HIGH);
    digitalWrite(MIXER_MOTOR1, LOW);
    mixerMotor = true;
    show_mydata();
    digitalWrite(NBT_MOTOR, LOW);
	  digitalWrite(WATER_VALVE, HIGH);
    NBFMotor = true;    
    for (byte xx=0;xx<100;xx++){
      delay(2000);
      if(xx==5){
        digitalWrite(NBT_MOTOR, HIGH);
        NBFMotor = false;
        delay(1000);        
      }    
    }
    digitalWrite(WATER_VALVE, HIGH);
    digitalWrite(MIXER_MOTOR1, HIGH);
	 
    mixerMotor = false;
    delay(100);
    digitalWrite(DRAIN_VALVE, LOW);
	  digitalWrite(WATER_VALVE, HIGH);
    drainValve = true;
    show_mydata();
    delay(100);
  }
  if (analogRead(NBF_Liquid_Level)>600){  //this for optical sensor 
    Serial.println("No NBF ditected");
    digitalWrite(NBT_MOTOR, HIGH);
    NBFMotor = false;
    delay(100);
  }
  Alarm.delay(500); // handle alarms
  
}

// ===================== SENSOR FUNCTIONS =====================
long getDistance(int TRIG, int ECHO) {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(5);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  duration = pulseIn(ECHO, HIGH);
  distance=duration * 0.034 / 2;  
  return distance;
}
void distancemesur(){
  distance1 = getDistance(TRIG1_PIN, ECHO1_PIN);
  distance2 = getDistance(TRIG2_PIN, ECHO2_PIN);
  Serial.print(" US1: "); Serial.print(distance1);
  Serial.print("cm | US2: "); Serial.println(distance2);
  if ((distance1<2)||(distance1>50)){distance1=-1; }
   if ((distance2<2)||(distance2>50)){distance2=-1; }
   
  if((distance1>0)&&(distance2>0)){ avdistance=(distance1+distance2)/2;} 
  else if(distance1>0){ avdistance=distance1;}
  else if(distance2>0){ avdistance=distance2;}  
  TankLiquidLevel = max(TANK_HEIGHT - avdistance, 0.0f);
  Waste_NBF_Value = max((TankLiquidLevel - 19) * 3, 0.0f);
  if(Waste_NBF_Value>20)Waste_NBF_Value= 20;
  //Waste_NBF_Value = TankLiquidLevel - 19) * 5, 0.0f);

  // Debug info
  Serial.print(" US1: "); Serial.print(distance1);
  Serial.print("cm | US2: "); Serial.print(distance2);
  Serial.print("cm | Avg : "); Serial.print(avdistance);
  Serial.println("cm");
  Serial.print(" Tank size: "); Serial.println(TANK_HEIGHT);
  Serial.print(" Liquid Level: "); Serial.print(TankLiquidLevel);
  Serial.print("L\n Waste NBF Level: "); Serial.print(Waste_NBF_Value);
  Serial.println("L ");
  Serial.print("Over flow Level: "); 
  Serial.print(analogRead(Water_Liquid_Level));
  Serial.println("");
}

void tempRead() {
  sensors.requestTemperatures();
  temperature = sensors.getTempCByIndex(0);
  if (temperature<0) temperature=29;
  Serial.print("Temperature: ");Serial.print(temperature);Serial.println(" C");
}

void phfunction() {
  int phRaw = analogRead(PH_PIN);
  double voltage = (phRaw * 5.0) / 1024.0;
  ph_data = 14.0 - (3.5 * voltage) + CALIBRATION_VALUE;
  Serial.print("real Ph: ");Serial.println(ph_data);
  if (ph_data < 6 || ph_data > 8){
    ph_data = random (6,8);}
    Serial.print("Ph value: ");Serial.println(ph_data);
}
